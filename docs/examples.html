Make a series

Replace the module contents in the Setup page with this code. It makes four stippled drawings with different seeds and adds four SVG download links.

import { PlotterEngine } from "https://cdn.jsdelivr.net/gh/seb-prjcts-be/vanilla.penplotter@v0.5.0/vanilla.penplotter.js";

for (let seed = 1; seed <= 4; seed += 1) {
  const plot = new PlotterEngine({ units: "mm", page: { width: 120, height: 80 } });
  plot.stipple([{ x: 12, y: 12 }, { x: 108, y: 12 },
    { x: 108, y: 68 }, { x: 12, y: 68 }],
    { count: 80, minDistance: 3, seed });
  const link = document.createElement("a");
  link.href = URL.createObjectURL(new Blob([plot.exportSVG()], { type: "image/svg+xml" }));
  link.download = `sheet-${seed}.svg`;
  link.textContent = `Save sheet ${seed}`;
  document.body.append(link, document.createElement("br"));
}
