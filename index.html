Vanilla first

No canvas capture.
No SVG detour.

The sketch creates plot geometry directly. p5.js and p5.waves are optional adapters, not foundations the engine depends on.

import { PlotterEngine } from "./vanilla.penplotter.js";

const plot = new PlotterEngine({
  units: "mm",
  page: { width: 210, height: 297 }
});

plot.circle(105, 145, 48);
plot.hatch(polygon, { spacing: 2.4 });

plot.optimize({ mergeTolerance: 0.05 });
const plan = plot.plan({ strategy: "nearest" });

console.log(plan.stats);
const hpgl = plot.exportHPGL();
